## Construct the report step by step
1. Version each case family and its expected source identity.
2. Run the same cases through the changed system.
3. Compare exact source decisions, including explicit no-source outcomes.
4. Keep individual failed case names so a developer can reproduce them.

## Separate timing from assertions
Use monotonic perf_counter around real operations. Capture queue waiting, provider duration, storage and whole-request latency when instrumenting the capstone. Retain timeout/error outcomes instead of calculating latency only for successful answers. Four fixture samples demonstrate the formula; they cannot support a stable production p95 claim.

## Evaluate changes under one budget
Compare a cached path, a larger model or a retrieval change on fixed cases, with a declared quality floor and resource allowance. Count failed attempts and retries in cost per accepted answer. If local usage counts are missing, report unknown token consumption and measure local resources separately.

## Semantic checks remain a different layer
Exact citation equality does not prove an answer is faithful to its citation. Use human-labeled claim support or a versioned model judge with recorded disagreement. Avoid calling a stochastic judge deterministic truth. Graduation needs both repeatable engineering checks and an honest semantic-quality scope.

```python
import statistics
import math

cases = [('supported','a1','a1'), ('missing',None,None), ('isolation',None,'b1'), ('empty',None,None)]
correct = sum(expected == observed for _,expected,observed in cases)
latency_ms = [10,12,20,80]
rank = max(0, math.ceil(.95*len(latency_ms))-1)
report = {'cases':len(cases),'source_accuracy':correct/len(cases),
          'p50_ms':statistics.median(latency_ms),'p95_ms':sorted(latency_ms)[rank],
          'failed':[name for name,expected,observed in cases if expected != observed],
          'measurement':'synthetic fixture, not provider timings'}
assert report['source_accuracy'] == .75
assert report['failed'] == ['isolation']
assert report['p95_ms'] == 80
print(report)
# Deterministic source checks do not certify that generated claims are faithful.
```

```python
import math
import statistics
import time

rows=[]
for name,critical,run in [('valid',False,lambda:'a1'),('outage',False,lambda:1/0),('owner-denied',True,lambda:None)]:
    started=time.perf_counter()
    try:

    except Exception as failure:passed=False;error=type(failure).__name__

print({'p50_ms':statistics.median(times),'p95_ms':times[math.ceil(.95*len(times))-1],
```

### evaluate.py

```python
"""Measured, versioned contract evaluation; no model-faithfulness certification."""
import argparse
import asyncio
import hashlib
import json
import math
import statistics
import tempfile
import time
from pathlib import Path
from pipeline import FixtureProvider, Store, answer_one

CASES = [
    {"id":"supported", "owner":"team-a", "query":"refund days?", "source":"a-refunds-v1", "critical":False},
    {"id":"punctuation", "owner":"team-a", "query":"refund!", "source":"a-refunds-v1", "critical":False},
    {"id":"missing", "owner":"team-a", "query":"salary", "source":None, "critical":False},
    {"id":"unauthorized", "owner":"team-b", "query":"refund days?", "source":None, "critical":True},
    {"id":"other-owner", "owner":"team-b", "query":"support hours", "source":"b-support-v1", "critical":False},
    {"id":"ambiguous", "owner":"team-a", "query":"how long?", "source":None, "critical":False},
]
DATASET_REVISION = "owner-source-cases-v2"

def summarize(rows, provider_identity):
    if not rows: raise ValueError("No observations")
    times = sorted(row["latency_ms"] for row in rows)
    failed = [row["case"] for row in rows if not row["passed"]]
    critical = [row["case"] for row in rows if row["critical"] and not row["passed"]]
    counts = [row["usage"] for row in rows if row["provider_called"]]
    known = bool(counts) and all(isinstance(u, dict) and type(u.get("input_tokens")) is int and type(u.get("output_tokens")) is int for u in counts)
    return {"dataset_revision":DATASET_REVISION,
        "dataset_sha256":hashlib.sha256(json.dumps(CASES, sort_keys=True).encode()).hexdigest(),
        "provider":provider_identity, "observations":len(rows), "source_accuracy":(len(rows)-len(failed))/len(rows),
        "failed_cases":failed, "critical_failures":critical,
        "p50_ms":statistics.median(times), "p95_ms":times[math.ceil(.95*len(times))-1],
        "latency_population":"every observed attempt, including errors; nearest-rank p95",
        "provider_calls":sum(row["provider_called"] for row in rows),
        "input_tokens":sum(u["input_tokens"] for u in counts) if known else None,
        "output_tokens":sum(u["output_tokens"] for u in counts) if known else None,
        "cost_per_accepted_result":None, "cost_note":"No billing tariff or local resource cost was measured; unknown is not zero.",
        "semantic_quality":"unmeasured; source identity is not claim support",
        "gate_passed":not critical and not failed, "rows":rows}

async def evaluate(provider, *, repeats=2):
    if type(repeats) is not int or not 1 <= repeats <= 100: raise ValueError("Use 1–100 repeats")
    rows = []
    with tempfile.TemporaryDirectory() as folder:
        store = Store(str(Path(folder)/"evaluation.sqlite3"))
        for _ in range(repeats):
            for case in CASES:
                started = time.perf_counter()
                try:
                    result = await answer_one(case["owner"], case["query"], provider, store)
                    row = {"case":case["id"], "passed":result["source"]==case["source"],
                        "critical":case["critical"], "error":None,
                        "provider_called":not result["cached"] and result["provider"]!="not-called",
                        "usage":result["usage"], "cached":result["cached"]}
                except Exception as error:
                    row = {"case":case["id"], "passed":False, "critical":case["critical"],
                        "error":type(error).__name__, "provider_called":True, "usage":None, "cached":False}
                row["latency_ms"] = (time.perf_counter()-started)*1000
                rows.append(row)
    return summarize(rows, provider.identity)

if __name__ == "__main__":
    parser = argparse.ArgumentParser()
    parser.add_argument("--repeats", type=int, default=2)
    parser.add_argument("--output", default="evaluation-observed.json")
    args = parser.parse_args()
    report = asyncio.run(evaluate(FixtureProvider(), repeats=args.repeats))
    Path(args.output).write_text(json.dumps(report, indent=2)+"\n")
    print(json.dumps({key:value for key,value in report.items() if key!="rows"}, indent=2))
    raise SystemExit(0 if report["gate_passed"] else 1)
```

### test_evaluate.py

```python
import unittest
from evaluate import evaluate, summarize
from pipeline import FixtureProvider, ProviderError

class EvaluationTests(unittest.IsolatedAsyncioTestCase):
    async def test_cold_and_warm_fixed_cases(self):
        report = await evaluate(FixtureProvider())
        self.assertEqual(report['observations'],12)
        self.assertTrue(report['gate_passed'])
        self.assertTrue(any(row['cached'] for row in report['rows']))
        self.assertIsNone(report['cost_per_accepted_result'])
        self.assertIsNone(report['input_tokens'])
    async def test_outages_remain_in_report(self):
        class Broken:
            identity='broken'
            async def answer(self,*args): raise ProviderError('synthetic unavailable')
        report=await evaluate(Broken(),repeats=1)
        self.assertFalse(report['gate_passed'])
        self.assertEqual(len(report['rows']),6)
        self.assertTrue(any(row['error']=='ProviderError' for row in report['rows']))
    async def test_critical_failure_is_not_hidden_by_average(self):
        report=await evaluate(FixtureProvider(),repeats=1)
        row=next(row for row in report['rows'] if row['case']=='unauthorized')
        row['passed']=False
        summary=summarize(report['rows'],'fixture')
        self.assertEqual(summary['critical_failures'],['unauthorized'])
        self.assertFalse(summary['gate_passed'])

if __name__=='__main__':unittest.main()
```
